interface GameItem {
  id: string;
  title: string;
  status: 'Jugando' | 'Completado' | 'No iniciado';
  hours: number;
  tags: string[];
}

const library: GameItem[] = [
  { id: '1', title: 'Cyberpunk 2077', status: 'Jugando', hours: 42, tags: ['RPG', 'Acción'] },
  { id: '2', title: 'Stardew Valley', status: 'Completado', hours: 120, tags: ['Simulación', 'Indie'] },
];

function filterByStatus(games: GameItem[], status: GameItem['status']) {
  return games.filter(game => game.status === status);
}

const playingGames = filterByStatus(library, 'Jugando');
console.log('Jugando:', playingGames);
